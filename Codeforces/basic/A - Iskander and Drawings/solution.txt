#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n;
    cin >> n;

    string s;
    cin >> s;

    int ans = 0, cnt = 0;

  for (int i = 0; i < n; i++) {
        if (s[i] == '#') {
            cnt++;
        } 
        else {
            ans = max(ans, (cnt + 1) / 2);
            cnt = 0;
        }
    }

    ans = max(ans, (cnt + 1) / 2);

    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
