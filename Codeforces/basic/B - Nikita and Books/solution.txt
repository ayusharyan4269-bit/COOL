#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n;
    cin >> n;

    ll a[n];

    for (int i = 0; i < n; i++) {
        cin >> a[i];
    }

    ll runningSum = 0;
    bool ok = true;

    for (int i = 0; i < n; i++) {
        runningSum += (a[i] - (i + 1));

        if (runningSum < 0) {
            ok = false;
        }
    }

    if (ok) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
