#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n, x, y;
    cin >> n >> x >> y;

    int p[n];

    for (int i = 0; i < n; i++) {
        cin >> p[i];
    }

    int g = __gcd(x, y);

    bool ok = true;

    for (int i = 0; i < n; i++) {
        if ((p[i] - (i + 1)) % g != 0) {
            ok = false;
        }
    }

    if (ok) {
        cout << "YES" << endl;
    } else {
        cout << "NO" << endl;
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
