#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

long long nCr(long long n, long long r) {
    if (r > n || r < 0)
        return 0;

    if (r > n - r)
        r = n - r;

    long long ans = 1;

    for (long long i = 1; i <= r; i++) {
        ans = ans * (n - i + 1) / i;
    }

    return ans;
}

long long ceilDiv(long long a, long long b) {
    return (a + b - 1) / b;
}

void solve() {
    ll n;
    cin >> n;

    ll c = 1;
    ll a = 0, b = 0;

    if (n % 2 == 0) {

        a = 2;
        b = n - 3;
    } else {
  
        ll m = (n - 1) / 2;
        if (m % 2 == 0) {
            a = m - 1;
            b = m + 1;
        } else {
            a = m - 2;
            b = m + 2;
        }
    }

    cout << a << " " << b << " " << c << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
