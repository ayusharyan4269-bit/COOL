#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    ll n, k, q;
    cin >> n >> k >> q;

    ll len = 0;
    ll ans = 0;

    for (int i = 0; i < n; i++) {
        ll x;
        cin >> x;

        if (x <= q) {
            len++;
        } else {
            if (len >= k) {
                ll t = len - k + 1;
                ans += t * (t + 1) / 2;
            }
            len = 0;
        }
    }

    if (len >= k) {
        ll t = len - k + 1;
        ans += t * (t + 1) / 2;
    }

    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
