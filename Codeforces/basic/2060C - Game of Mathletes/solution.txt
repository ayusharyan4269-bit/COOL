#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n,k;
    cin>>n>>k;
int arr[n];
    for (int i = 0; i < n; i++) {
        cin>>arr[i];
    }

sort (arr , arr + n);
int score =0;
int a = 0;
int b = n-1;

while (a < b) {
    if (arr[a] + arr[b] < k) {
        a++;
    }
    else if (arr[a] + arr[b] > k) {
        b--;
    }
    else {
        score++;
        a++;
        b--;
    }
}

cout<<score<<endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
