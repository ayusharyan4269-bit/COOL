#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

long long nCr(long long n, long long r) {
    if (r > n || r < 0)
        return 0;

    if (r > n - r)
        r = n - r;

    long long ans = 1;

    for (long long i = 1; i <= r; i++) {
        ans = ans * (n - i + 1) / i;
    }

    return ans;
}

long long ceilDiv(long long a, long long b) {
    return (a + b - 1) / b;
}

void solve() {
    int n;

int x=0;

    if (n==2 || n==3 )

 x=n;

x = n % 2;
    }

}

        solve();
    }

    return 0;
}
