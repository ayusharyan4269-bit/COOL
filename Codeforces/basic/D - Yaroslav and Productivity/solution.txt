#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n, m;
    cin >> n >> m;

    ll pref[n + 1];
    pref[0] = 0;

    for (int i = 1; i <= n; i++) {
        cin >> pref[i];
        pref[i] += pref[i - 1];
    }

    int b[m];

    for (int i = 0; i < m; i++) {
        cin >> b[i];
    }

    sort(b, b + m);

    ll ans = 0;
    int last = 0;

    for (int i = 0; i < m; i++) {
        ans += abs(pref[b[i]] - pref[last]);
        last = b[i];
    }

    ans += pref[n] - pref[last];

    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
