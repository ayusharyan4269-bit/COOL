#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

void solve() {
    int n;
    cin >> n;

    string s;
    cin >> s;

    ll ans = 0;
    ll cntB = 0;

    for (int i = n - 1; i >= 0; i--) {
        if (s[i] == 'B') {
            cntB++;
        } else {
            if (cntB > 0) {
                ans += cntB;
                cntB = 1;
            }
        }
    }

    cout << ans << endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
