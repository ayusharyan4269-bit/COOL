#include <bits/stdc++.h>
using namespace std;

#define ll long long
#define endl '\n'

long long nCr(long long n, long long r) {
    if (r > n || r < 0)
        return 0;

    if (r > n - r)
        r = n - r;

    long long ans = 1;

    for (long long i = 1; i <= r; i++) {
        ans = ans * (n - i + 1) / i;
    }

    return ans;
}

long long ceilDiv(long long a, long long b) {
    return (a + b - 1) / b;
}

void solve() {
    int n;
    cin>>n;
    ll sum = 0;
    ll mx = 0;
    int ans = 0;

    for (int i = 0; i < n; i++) {
        ll x;
        cin >> x;

        sum += x;
        mx = max(mx, x);

        if (2 * mx == sum)
            ans++;
    }

    cout << ans << endl;
}


int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    int t;
    cin >> t;

    while (t--) {
        solve();
    }

    return 0;
}
