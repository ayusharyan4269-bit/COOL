#include <bits/stdc++.h>
using namespace std;

void solve() {
    string str;
    cin >> str;

    string target = "hello";
    int j = 0;

    for (int i = 0; i < str.size(); i++) {
        if (str[i] == target[j]) {
            j++;
            if (j == target.size())
                break;
        }
    }

    if (j == 5)
        cout << "YES"<<endl;
    else
        cout << "NO"<<endl;
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(nullptr);

    solve();

    return 0;
}
